# 原先的世界地图代码
```
<template>
  <div class="world-map-wrapper">
    <h2 class="global-title">
      <i class="iconfont icon-diqiu icon-color"></i>
      Global Server Distribution
    </h2>  

    <!-- 世界地图 -->
    <div ref="chart" class="chart-container"></div>

    <div class="stats-tabs">
      <div :class="['tab-item', { 'is-active': hasUserClicked && activeTabIndex === -1 }]" @click="selectTab(-1)">
        <div class="tab-title">All Sites</div>
        <div class="tab-stats-row">
          <div class="tab-value-box">
            Total: <span>{{ summaryData.total_servers }}</span>
          </div>
        </div>
      </div>

      <div :class="['tab-item', { 'is-active': hasUserClicked && activeTabIndex === index }]"
        v-for="(site, index) in siteStats" 
        :key="site.name" 
        @click="selectTab(index)"
      >
        <div class="tab-title">{{ site.name }}</div>
        <div class="tab-stats-row">
          <div class="tab-value-box">
            <span class="color-green">{{ site.online }} Online</span> 
          </div>
          <div class="tab-value-box">
            <span class="color-red">{{ site.offline }} Offline</span>
          </div>
        </div>
      </div>
    </div>

    <div class="large-stats-container">
      <div class="large-tab">
        <div class="large-label">Total Servers</div>
        <div class="large-num color-blue">{{ displayStats.total }}</div>
      </div>
      <div class="large-tab">
        <div class="large-label">Online Servers</div>
        <div class="large-num color-green">{{ displayStats.online }}</div>
      </div>
      <div class="large-tab">
        <div class="large-label">Offline Servers</div>
        <div class="large-num color-red">{{ displayStats.offline }}</div>
      </div>
      <div class="large-tab">
        <div class="large-label">High Load Servers</div>
        <div class="large-num color-orange">{{ displayStats.highLoad }}</div>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, onMounted, onBeforeUnmount } from 'vue';
import * as echarts from 'echarts';
import { worldMapList } from '@/api/dashboard/echartList';
import { siteDataStore } from '@/stores/dashboard/dashboard';
import { useRoute } from 'vue-router';
import { useDocumentVisibility } from '@vueuse/core';

const route = useRoute();
const store = siteDataStore();
const chart = ref(null);
let chartInstance: any = null;
const activeTabIndex = ref(-1);
const hasUserClicked = ref(false);
let timer: any = null; // 1. 定义定时器变量
let visibility = useDocumentVisibility();

// 响应式统计数据
const summaryData = ref({
  total_servers: 0,
  online_servers: 0, 
  offline_servers: 0, 
  high_load_servers: 0
});
const siteStats = ref<any[]>([]);

const colorPalette = [
  '#60a5fa', '#10b981', '#f59e0b', '#8b5cf6', '#ef4444', 
  '#06b6d4', '#ec4899', '#f97316', '#84cc16'
];

// 坐标：Key 对应 API 返回的 site.name
const geoCoordMap: Record<string, [number, number]> = {
  'FVN': [108.2772, 14.0583],   // Vietnam
  'FJZ': [-106.4800, 31.7300],  // Ciudad Juárez, Mexico
  'FOL': [114.0579, 22.5431],   // Shenzhen
  'FSJ': [-113.9071, 36.4173],  // SanJosé
  'FTX': [-89.6165, 43.7844],   // Texas
  'FCZ': [14.42, 50.08],        // Czechia
  'FGP': [-103.3500, 20.6600]   // Guadalajara, Mexico
};

const displayStats = computed(() => {
  if (activeTabIndex.value === -1) {
    return {
      total: summaryData.value.total_servers,
      online: summaryData.value.online_servers,
      offline: summaryData.value.offline_servers, 
      highLoad: summaryData.value.high_load_servers
    };
  } 
  // 如果选中的是具体的某个 Site
  const currentSite = siteStats.value[activeTabIndex.value];
  return {
    total: currentSite?.total || 0,
    online: currentSite?.online || 0,
    offline: currentSite?.offline || 0,
    highLoad: currentSite?.high_load || 0
  };
});

watch(visibility, (newVal) => {
  if (newVal === 'visible') {
    startPolling();
  } else {
    stopPolling();
  }
});

const getBasePath = () => {
  return window.location.hostname === 'ite.cns.myfiinet.com' ? '/genius-cloud' : '';
};

// 获取数据并触发渲染
const fetchMapData = async () => {
  try {
    const res: any = await worldMapList();
    if (res.status && res.data?.payload?.data) {
      let { summary, sites } = res.data.payload.data;
      summaryData.value = summary; 
      siteStats.value = (sites || []); 
    }
  } catch (error) {
    console.error('获取地图数据失败:', error);
  }
};

// 核心渲染逻辑：完全由 siteStats 驱动
const renderOption = () => {
  let colorsArr: any = {};
  if (!chartInstance) return;
  const seriesData = siteStats.value
    .filter((site) => geoCoordMap[site.name]) 
    .map((site, index) => {
      // 检查当前点是否被 Tab 选中
      const isSelected = activeTabIndex.value !== -1 && siteStats.value[activeTabIndex.value].name === site.name;
      const baseColor = colorPalette[index % colorPalette.length];
      colorsArr[site.name] = baseColor; // 存储颜色映射
      return {
        name: site.name,
        site: site.name,
        value: [...geoCoordMap[site.name], 100], // [经度, 纬度, 值]
        isSelected: isSelected, 
        itemStyle: {
          color: baseColor, // 使用匹配到的颜色
          opacity: (activeTabIndex.value === -1 || isSelected) ? 1 : 0.3
        }
      };
    });
  siteDataStore().setBarColors(colorsArr); // 同步颜色到全局 store
  chartInstance.setOption({
    animation: true,
    geo: {
      map: 'world',
      roam: true,
      zoom: 1.2,
      center: [16, 16],
      itemStyle: { areaColor: '#4f46e5', borderColor: '#111' }
    },
    emphasis: {
      itemStyle: {
        areaColor: '#6366f1' // 鼠标悬停时，陆地板块的颜色
      },
      label: {
        show: false
      }
    },
    series: [
      {
        name: 'Sites',
        type: 'effectScatter',
        coordinateSystem: 'geo',
        // 基础数据配置
        data: seriesData.map((item) => ({
          ...item,
          symbolSize: item.isSelected ? 24 : 12, 
          rippleEffect: {
            period: 4,
            scale: item.isSelected ? 2.8 : 2.5,
            brushType: 'stroke', 
            color: item.itemStyle.color
          }
        })),
        showEffectOn: 'render',
        
        // 鼠标移动上去（Hover）的联动效果
        emphasis: {
          disabled: false,
          scale: true,
          itemStyle: {
            shadowBlur: 10
          }
        },
        label: {
          show: true,
          position: 'right',
          formatter: '{b}',
          color: '#ffffff'
        }
      }
    ],
    tooltip: {
      show: true,
      trigger: 'item',
      backgroundColor: '#0f172a',
      borderColor: '#334155',
      textStyle: {
        color: '#ffffff' 
      },
      formatter: (params: any) => {
        const stats = siteStats.value.find((s) => s.name === params.data.site);
        if (!stats) return params.name;
        const total = stats.total;
        const highLoad = stats.high_load || 0;
        return `
          <div style="min-width: 150px;">
            <div style="font-weight: bold; font-size: 16px; color: #ffffff; margin-bottom: 5px;">${stats.name}</div>
            <hr style="border: none; border-top: 1px solid var(--theme-border-color); margin: 5px 0;" />
            <div style="color: #10b981; margin-bottom: 3px;">Online: <span style="font-weight: bold;">${stats.online}</span></div>
            <div style="color: #ef4444; margin-bottom: 3px;">Offline: <span style="font-weight: bold;">${stats.offline}</span></div>
            <div style="color: #f59e0b; margin-bottom: 3px;">High Load: <span style="font-weight: bold;">${highLoad}</span></div>
            <div style="color: #ffffff; margin-bottom: 3px;">Total: <span style="font-weight: bold;">${total}</span></div>
            <hr style="border: none; border-top: 1px solid var(--theme-border-color); margin: 5px 0;" />
            <div style="color: #60a5fa;">Availability: <span style="font-weight: bold;">${stats.availability}</span></div>
          </div>
        `;
      }
    }
  });
};

const selectTab = async (index: number) => {
  hasUserClicked.value = true;
  activeTabIndex.value = index;
  const selectedSite = index !== -1 ? siteStats.value[index].name : 'All Sites';
  store.setSite(selectedSite);
  await fetchMapData();
  renderOption();
};

const initChart = () => {
  if (chart.value) {
    chartInstance = echarts.init(chart.value);
    const basePath = getBasePath(  );
    fetch(`${basePath}/world.json`)
      .then((res) => res.json())
      .then((worldGeoJson) => {
        echarts.registerMap('world', worldGeoJson);
        renderOption();
        chartInstance.on('click', async (params: any) => {
          if (params.componentType === 'series') {
            hasUserClicked.value = true;
            const siteName = params.data.site;
            const idx = siteStats.value.findIndex((s) => s.name === siteName);
            activeTabIndex.value = idx;
            store.setSite(siteName);
            await fetchMapData();
            renderOption(); 
          }
        });
      });
  }
};

const startPolling = () => {
  stopPolling(); 
  timer = setInterval(async () => {
    console.log('Refreshing map data...');
    await fetchMapData();
    renderOption(); 
  }, 30000); 
};

const stopPolling = () => {
  if (timer) {
    clearInterval(timer);
    timer = null;
  }
};

onMounted(async () => {
  await fetchMapData();
  initChart();
  startPolling();
  window.addEventListener('resize', () => chartInstance?.resize());
  setTimeout(() => {
    if (!route.query.site) return;
    const index = siteStats.value.findIndex((data) => route.query.site === data.name);
    selectTab(index);
  }, 300);
});

onBeforeUnmount(() => {
  stopPolling();
  chartInstance?.dispose();
  window.removeEventListener('resize', () => chartInstance?.resize());
});
</script>
```
