8. 会议纪要: 新建模式提供议程与会议记录输入及保存操作；历史模式仅允许查看。

import { siteBuMixin } from '@/utils/mixins';

import ActionDetailDialog from '@project_sensing/components/workspace/meeting/dialogs/ActionDetailDialog.vue';
import ActionHistoryDialog from '@project_sensing/components/workspace/meeting/dialogs/ActionHistoryDialog.vue';
import ActionItemDialog from '@project_sensing/components/workspace/meeting/dialogs/ActionItemDialog.vue';
import MeetingSidebar from '@project_sensing/components/workspace/meeting/MeetingSidebar.vue';
import { DatePicker } from 'element-ui';
import locale from 'element-ui/lib/locale';
import lang from 'element-ui/lib/locale/lang/en';
import moment from 'moment-timezone';
import { formatOwnerName } from '@project_sensing/utils/utils.js';

locale.use(lang);

export default {
  name: 'MeetingMinutes',
  mixins: [siteBuMixin],
  components: {
    elDatePicker: DatePicker,
    PageFitLayout,
    MeetingSidebar,
    ActionItemRow,
    ActionItemDialog,
    ActionDetailDialog,
    ActionHistoryDialog,
  },
  props: {
    currentUser: {
      type: Object,
      default: () => ({}),
    },
    members: {
      type: Array,
      default: () => [],
    },
  },

  data () {
    return {
      // 页面模式：'create' 可编辑（新建），'view' 只读（历史会议 / 默认）
      pageMode: 'view',
      // 空表单模板：新建 / 重置时基于此返回副本
      emptyMeetingForm: {
        subject: null,
        meeting_time: null,
        location: null,
        host_english_name: null,
        host_chinese_name: null,
        host_id: null,
        attendees: [],
        action_items: [],
        meeting_content: null,
        project_id: null,
        meeting_status: 'save',
      },
      actionDialogVisible: false,
      actionDetailDialogVisible: false,
      historyDialogVisible: false,
      selectedActionId: null,
      selectedMeetingIndex: null,
      selectedMeetingId: null,
      isNewMeeting: false,
      // 右侧内容区唯一数据源，create / view 共用
      meetingForm: {
        subject: null,
        meeting_time: null,
        location: null,
        host_english_name: null,
        host_chinese_name: null,
        host_id: null,
        attendees: [],
        action_items: [],
        meeting_content: null,
        project_id: null,
        meeting_status: 'submit',
      },
      // 本地新增的待办（未入库），保存会议时随会议一起提交
      localActionItems: [],
      // 当前正在编辑的本地新增待办索引，-1 表示不是本地新增
      editingLocalIndex: -1,
      // 详情弹窗数据源（由列表行传入）
      actionDetailForm: {},
      saveActionLoading: false,
      minutesLoading: false,
      // 待办 Owner 可选项
      actionOwners: [],
      // 待办历史
      actionItemsHistory: {
        actions: [],
        loading: false,
      },
      // 会议列表
      projectMeetings: [],
      // 表单的必填规则，绑定在 Meeting Name 和 Location上
      requiredRules: [
        value => !!value || 'This field is required',
      ],
      formErrors: {
        meeting_time: '',
      },
      closeActionConfirmVisible: false,
      closeActionLoading: false,
      // 当前要处理哪一条数据
      pendingCloseAction: null,
      // 最近一次成功保存后的会议内容快照。
      savedMeetingSnapshot: null,
      // 保存中状态；避免保存请求过程中触发离开提示。
      savingMeetingStatus: null,
      deleteMeetingDialogVisible: false,
      deletingMeeting: null,
      deletingMeetingId: null,
    };
  },

  computed: {
    // 是否新建（可编辑）模式
    isCreateMode () {
      return this.pageMode === 'create';
    },
    // 是否只读（历史 / 默认）模式
    isViewMode () {
      return this.pageMode === 'view';
    },
    // 新建会议或 save 草稿可编辑；submit 会议只读。
    // canEditMeeting () {
    //   if (!this.hasProjectMember) return false;
    //   return this.isCreateMode || this.meetingForm.meeting_status === 'save';
    // },
    /**
     * 可编辑 Meeting Name、Date、Location 的条件：
     * 1. New Meeting；
     * 2. Draft（meeting_status = save）；
     * 3. Meeting List 第一条且已提交的会议。
     *
     * 其他已提交的历史会议维持只读。
     */
    canEditMeeting () {
      if (!this.hasProjectMember) return false;
      return this.isCreateMode
        || this.meetingForm.meeting_status === 'save'
        || (
          this.meetingForm.meeting_status === 'submit'
          && this.isLatestMeeting
        );
    },
    requestParams () {
      return {
        site: this.siteBuObj.site,
        customer: this.siteBuObj.customer,
        project_id: this.$route.query.projectId,
      };
    },
    // 看看是不是项目组成员，是项目组成员可以创建会议
    hasProjectMember () {
      return this.hasPermission(this.currentUser, this.members);
    },
    // 用于显示附件上传人：工号 + 姓名
    currentUserDisplayName () {
      const user = this.currentUser || {};
      const employeeNo = user.id;
      const employeeName = user.name;
      if (employeeNo && employeeName) {
        return `${employeeNo} ${employeeName}`;
      }
      return employeeNo || employeeName || 'Current User';
    },
    // 只有当前内容与初始 / 最近保存快照不一致时才提示。
    hasUnsavedMeeting () {
      if (!this.savedMeetingSnapshot) return false;
      // New Meeting、旧 Draft、或最新已提交会议，
      // 只要允许编辑且内容变化，就提示保存。
      return this.canEditMinutes
        && this.getMeetingEditSignature() !== this.savedMeetingSnapshot;
    },
    //  Meeting List 第一笔为最新会议，使用 ID 判断，避免 selectedMeetingIndex 与列表组件同步时机导致判断错误。
    isLatestMeeting () {
      const latestMeeting = this.projectMeetings[0];
      return !this.isCreateMode
        && !!latestMeeting
        && String(latestMeeting.id) === String(this.selectedMeetingId);
    },
    // 新建会议、草稿会议，以及最新已提交会议的参会状态可编辑。
    canEditAttendees () {
      return this.canEditMeeting || (
        this.hasProjectMember
        && this.meetingForm.meeting_status === 'submit'
        && this.isLatestMeeting
      );
    },
    // 新建会议、旧 Draft，或最新一笔已提交会议，允许编辑 Meeting Minutes。
    canEditMinutes () {
      return this.canEditMeeting || (
        this.hasProjectMember
        && this.meetingForm.meeting_status === 'submit'
        && this.isLatestMeeting
      );
    },
    // 只有当前用户在项目成员列表中的角色为 PM 时，才具有删除 Meeting Minutes 的权限。
    isProjectPm () {
      return this.members.some(member => (
        String(member.user_id) === String(this.currentUser.id)
        && String(member.project_role || '').toUpperCase() === 'PM'
      ));
    },
  },

  watch: {
    requestParams: {
      immediate: true,
      deep: true,
      handler (params) {
        if (Object.values(params).every(Boolean)) {
          this.getMeetingList().catch(() => {});
        }
      },
    },
  },

  // 离开当前路由前检查 New Meeting 是否尚未保存。
  beforeRouteLeave (to, from, next) {
    if (!this.hasUnsavedMeeting || this.savingMeetingStatus) {
      next();
      return;
    }
    document.body.classList.add('has-unsaved-meeting-modal');
    this.$Modal.confirm({
      title: 'Unsaved Meeting',
      content: 'You have an unsaved meeting. Are you sure you want to leave this page?',
      okText: 'Leave',
      cancelText: 'Stay',
      iconType: 'ios-alert',
      width: 480,
      onOk: () => {
        this.removeUnsavedMeetingModalClass();
        next();
      },
      onCancel: () => {
        this.removeUnsavedMeetingModalClass();
        next(false);
      },
    });
  },

  methods: {
    formatOwnerName,
    // 根据创建顺序来排序
    sortMembersByCreatedAt (members = []) {
      return [...members].sort((a, b) => {
        const aTime = new Date(a.created_at || 0).getTime();
        const bTime = new Date(b.created_at || 0).getTime();

        // 项目成员建立时间早的在前；新增加的成员排在后。
        if (aTime !== bTime) return aTime - bTime;

        // created_at 相同或异常时，用成员记录 id 保证排序稳定。
        return Number(a.id || 0) - Number(b.id || 0);
      });
    },

    openDeleteMeetingDialog (meeting) {
      if (!this.hasProjectMember || !meeting || !meeting.id) return;
      this.deletingMeeting = meeting;
      this.deleteMeetingDialogVisible = true;
    },
    cancelDeleteMeeting () {
      if (this.deletingMeetingId !== null) return;
      this.deleteMeetingDialogVisible = false;
      this.deletingMeeting = null;
    },
    async confirmDeleteMeeting () {
      const meeting = this.deletingMeeting;
      console.log('meeting', meeting);
      if (!this.hasProjectMember || !meeting || !meeting.id || this.deletingMeetingId !== null) return;
      this.deletingMeetingId = meeting.id;
      try {
        const { data: res } = await meetingMinutes.deleteMeeting({
          customer: this.siteBuObj.customer,
          site: this.siteBuObj.site,
          id: meeting.id,
        });
        if (!res || !res.status) throw new Error((res && res.message) || 'Failed to delete meeting minutes');

        const deletedMeetingId = String(meeting.id);
        this.deleteMeetingDialogVisible = false;
        this.deletingMeeting = null;

        // 删除当前正在查看的会议时，清空右侧详情，避免继续显示已删除资料。
        if (String(this.selectedMeetingId) === deletedMeetingId) {
          this.selectedMeetingId = null;
          this.selectedMeetingIndex = null;
          this.pageMode = 'view';
          this.isNewMeeting = false;
          this.localActionItems = [];
          this.meetingForm = this.getEmptyMeetingForm(this.requestParams.project_id);
          this.savedMeetingSnapshot = null;
        }
        await this.getMeetingList();
        this.showMessage('success', 'Delete meeting minutes successfully');
      } catch (error) {
        console.error(error);
        this.showMessage('error', error.message || 'Failed to delete meeting minutes');
      } finally {

    // 将后端 ISO 时间格式统一为 el-date-picker 的 value-format
    normalizeMeetingTime (value) {
      if (!value) return null;
      const parsed = moment(value);
      if (!parsed.isValid()) {
        console.warn('[MeetingMinutes] Invalid meeting_time:', value);
        return null;
      }
      // 必须与 value-format="yyyy-MM-dd HH:mm:ss" 一致
      return parsed.format('YYYY-MM-DD HH:mm:ss');

      if (id === null || id === undefined || id === '') return null;

    /**
     * get-project-meeting 的 meeting_status 参数用于筛选。
     * 分别请求 save 和 submit，再合并成会议列表。
     */
    async getMeetingList () {
      const [saveResponse, submitResponse] = await Promise.all([
        meetingMinutes.getMeetings({
          ...this.requestParams,
          meeting_status: 'save',
        }),
        meetingMinutes.getMeetings({
          ...this.requestParams,
          meeting_status: 'submit',
        }),
      ]);
      const saveResult = saveResponse && saveResponse.data;
      const submitResult = submitResponse && submitResponse.data;
      if (!saveResult || !saveResult.status) {
        throw new Error((saveResult && saveResult.message) || 'Failed to get saved meetings');
      }
      if (!submitResult || !submitResult.status) {
        throw new Error((submitResult && submitResult.message) || 'Failed to get submitted meetings');
      }
      const meetingsById = new Map();
      [...(submitResult.data || []), ...(saveResult.data || [])].forEach(meeting => {
        meetingsById.set(meeting.id, {
          ...meeting,
          meeting_status: meeting.meeting_status || 'submit',
        });
      });
      const getMeetingSortTime = meeting => {
        const value = meeting.updated_at || meeting.created_at || meeting.meeting_time;
        const time = value ? new Date(value).getTime() : 0;
        return Number.isNaN(time) ? 0 : time;
      };
      this.projectMeetings = Array.from(meetingsById.values())
        .sort((a, b) => getMeetingSortTime(b) - getMeetingSortTime(a));
      return this.projectMeetings;
    },

          ...this.localActionItems[this.editingLocalIndex],
          ...detailForm,
          // 同步删除后的附件展示列表

          // 同步删除后的待上传文件列表
          pendingAttachmentFiles,
        });
        this.actionDetailDialogVisible = false;
        return;
      }
      // 后端已经存在的 Action Item。
      const actionItem = this.meetingForm.action_items.find(
        item => String(item.id) === String(this.selectedActionId),
      );
      if (!actionItem) return;
      this.saveActionLoading = true;
      try {
        const formData = new FormData();
        const originalFileList = Array.isArray(actionItem.file_list)
          ? actionItem.file_list
          : [];
        // ActionDetailDialog 删除附件后回传的是 attachments。
        const currentAttachments = Array.isArray(detailForm.attachments)
          ? detailForm.attachments
          : [];
        // 根据 URL / 文件名判断“旧附件是否仍存在”，
        // 不再仅依赖附件数组长度。
        const getAttachmentKey = attachment => (
          attachment.file_url
          || attachment.url
          || attachment.file_name
          || attachment.name
          || ''
        );
        const remainingAttachmentKeys = new Set(
          currentAttachments.filter(attachment => !attachment.isLocal).map(getAttachmentKey),
        );
        const hasFileDeleted = originalFileList.some(
          attachment => !remainingAttachmentKeys.has(getAttachmentKey(attachment)),
        );

        const updatedData = {
          ...this.getChangedFields(actionItem, detailForm),
          id: actionItem.id,
          project_id: this.requestParams.project_id,
          meeting_id: actionItem.meeting_id,
        };
        // 用户删除已有附件时，将“剩余文件列表”传给后端。

            .filter(attachment => !attachment.isLocal)

              file_name: attachment.file_name || attachment.name,
              file_url: attachment.file_url || attachment.url,
              upload_by: attachment.upload_by || attachment.uploadedBy,
            }));

    /**
     * 保存会议
     * Save：传 meeting_status=save，草稿仍可编辑。
     * Submit：传 meeting_status=submit，提交后页面变为只读。
     */
    async saveMeeting (targetStatus = 'save') {
      if (!['save', 'submit'].includes(targetStatus)) return;
      if (!this.canEditMinutes || this.savingMeetingStatus) return;

      this.formErrors.meeting_time = this.meetingForm.meeting_time ? '' : 'This field is required';
      // const isFormValid = this.$refs.meetingForm ? this.$refs.meetingForm.validate() : false;
      // 在发起请求前记录模式，后面不要再依赖已变化的 pageMode 判断。
      const creating = this.isCreateMode;
      // 新建、草稿、最新已提交会议在可编辑时都会渲染 v-form，只要表单存在，就校验 Meeting Name 与 Location。
      const isFormValid = this.$refs.meetingForm
        ? this.$refs.meetingForm.validate()
        : true;

      if (!isFormValid || !this.meetingForm.meeting_time) return;
      this.savingMeetingStatus = targetStatus;

      try {
        // 创建前先记录已有 ID，用于创建接口没有返回 ID 时定位新记录。
        const existingIds = new Set(this.projectMeetings.map(meeting => String(meeting.id)));
        const payload = {
          ...this.meetingForm,
          project_id: this.requestParams.project_id,
          meeting_status: targetStatus,
        };
        // if (!creating) delete payload.attendees;
        // 仅允许编辑最新会议/草稿的 Attendance Check；其他场景不应把 attendees 送到更新接口
        if (!this.canEditAttendees) delete payload.attendees;
        // 不再随会议接口传递 action_items
        delete payload.action_items;
        let response;
        if (creating) {
          response = await meetingMinutes.createMeeting(payload, this.requestParams);
        } else {
          // 防止 selectedMeetingId 为空时发出缺少 id 的 PATCH。
          if (!this.selectedMeetingId) {
            throw new Error('当前会议缺少Id，无法更新');
          }
          response = await meetingMinutes.updateMeeting(
            {
              ...payload,
              id: this.selectedMeetingId,
            },
            this.requestParams,
          );
        }
        const res = response && response.data;
        if (!res || !res.status) {
          throw new Error((res && res.message) || 'Save meeting failed');
        }
        const responseData = res.data || {};
        let meetingId = this.normalizeMeetingId(responseData.id);
        // 创建接口未返回 ID 时，从刷新后的列表中找刚创建的记录。
        if (creating && !meetingId) {
          await this.getMeetingList();
          const targetDate = this.meetingForm.meeting_time
            ? moment(this.meetingForm.meeting_time).format('YYYY-MM-DD')
            : '';
          const newlyCreatedMatches = this.projectMeetings.filter(meeting => {
            if (existingIds.has(String(meeting.id))) return false;
            const listDate = meeting.meeting_time
              ? moment(meeting.meeting_time).format('YYYY-MM-DD')
              : '';
            return meeting.subject === this.meetingForm.subject && meeting.location === this.meetingForm.location && String(meeting.host_id || '') === String(this.meetingForm.host_id || '') && targetDate === listDate;
          });
          if (newlyCreatedMatches.length) {
            // getMeetingList 已按更新时间倒序，优先取最新匹配记录。
            meetingId = newlyCreatedMatches[0].id;
          }
        }
        // 创建成功但无法确认 ID 时，不能切换到 update 模式。
        if (creating && !meetingId) {
          await this.getMeetingList();
          throw new Error('会议已创建，但前端没有取得会议 ID, 无法继续更新。请确认 createMeeting 响应返回 id 或 meeting_id');
        }
        // 无论是首次 Create，还是已有 Draft / Meeting 的 Update，
        // 后续都统一使用当前有效的会议 ID。
        const resolvedMeetingId = this.normalizeMeetingId(meetingId || this.selectedMeetingId);
        if (!resolvedMeetingId) {
          throw new Error('Meeting saved, but meeting ID is missing.');
        }
        this.selectedMeetingId = resolvedMeetingId;
        this.meetingForm.meeting_status = targetStatus;
        // Save 与 Submit 都保存本地新增的 Action Items
        // 区别仅由 targetStatus 决定 meeting_status 传 save 或 submit
        await this.savePendingActionItems(resolvedMeetingId);
        // Save 后保留草稿编辑能力；Submit 后 canEditMeeting 会变为 false
        this.pageMode = 'view';
        this.localActionItems = [];
        // 后续如果用户继续修改 Draft，computed.hasUnsavedMeeting 会变成 true
        this.savedMeetingSnapshot = this.getMeetingEditSignature();
        this.showMessage('success', targetStatus === 'save' ? 'Save meeting successfully' : 'Submit meeting successfully');
        await this.getMeetingList();
        const savedMeetingIndex = this.projectMeetings.findIndex(
          meeting => String(meeting.id) === String(this.selectedMeetingId),
        );
        // 刚创建并保存成功的会议，视为最新会议，固定放到列表第一笔。
        // 因此 Agenda & Meeting Minutes 会保持可编辑，Save 按钮也会显示。
        if (creating && savedMeetingIndex > 0) {
          const [savedMeeting] = this.projectMeetings.splice(savedMeetingIndex, 1);
          this.projectMeetings.unshift(savedMeeting);
        }
        this.selectedMeetingIndex = this.projectMeetings.findIndex(
          meeting => String(meeting.id) === String(this.selectedMeetingId),
        );
        await this.getOpenActions();
      } catch (err) {
        console.error(err);
        this.showMessage('error', err.message || 'Save meeting failed');
      } finally {
        this.savingMeetingStatus = null;
      }
    },
    async savePendingActionItems (meetingId) {
      const normalizedMeetingId = this.normalizeMeetingId(meetingId);
      const pendingItems = Array.isArray(this.localActionItems) ? this.localActionItems : [];
      if (!normalizedMeetingId || !pendingItems.length) return;
      await Promise.all(
        pendingItems.map(async item => {
          const actionItemData = {
            ...item,
            meeting_id: normalizedMeetingId,
            project_id: this.requestParams.project_id,
          };
          const pendingAttachmentFiles = (
            actionItemData.pendingAttachmentFiles || []
          );
          delete actionItemData.pendingAttachmentFiles;
          delete actionItemData.localAttachments;
          delete actionItemData.attachments;
          delete actionItemData.file_list;
          delete actionItemData.file_num;
          const formData = new FormData();
          formData.append('data', JSON.stringify(actionItemData));
          formData.append('project_id', String(this.requestParams.project_id));
          formData.append('meeting_id', normalizedMeetingId);
          pendingAttachmentFiles.forEach(file => {
            formData.append('files', file);
          });
          const { data: res } = await meetingMinutes.addMeetingActionV2(formData, this.requestParams);
          if (!res || !res.status) {
            throw new Error(
              (res && res.message)
              || 'Failed to save Action Item',
            );
          }
        }),
      );
    },
    getLastMinutesData () {
      // 已有内容时先确认，避免覆盖用户已填写的纪要
      if (this.meetingForm.meeting_content && this.meetingForm.meeting_content.trim()) {
        document.body.classList.add('has-unsaved-meeting-modal');
        this.$Modal.confirm({
          title: 'Overwrite Existing Minutes?',
          content: 'Meeting minutes already have content. Continuing will overwrite the existing content. Are you sure you want to continue?',
          okText: 'Continue',
          cancelText: 'Cancel',
          iconType: 'ios-alert',
          width: 480,
          onOk: () => {
            this.loadLastMinutes();
            document.body.classList.remove('has-unsaved-meeting-modal');
          },
        });
        return;
      }
      this.loadLastMinutes();
    },
    // 实际请求逻辑
    loadLastMinutes () {
      this.minutesLoading = true;
      meetingMinutes.getLastMinutes(this.requestParams)
        .then(({ data: res }) => {
          const { status, message, data } = res;
          if (status) {
            this.meetingForm.meeting_content = data;
          } else {
            this.$Message.error(message);
          }
        })
        .catch(err => {
          this.$Message.error(err.message || 'Failed to load latest minutes');
          console.error(err);
        })
        .finally(() => {
          this.minutesLoading = false;
        });
    },
    showMessage (type, message) {
      if (this.$Message && this.$Message[type]) {
        this.$Message[type](message);
      }
    },
    // 判断当前用户有没有操作权限，必须要属于这个项目的成员，编辑 Action Item 也是
    hasPermission (user, data) {
      if (!user || !user.id || !Array.isArray(data)) return false;
      return data.some(
        item => String(item.user_id) === String(user.id),
      );
    },
    // ActionDetailDialog 顶部 Close 图标：复用 ActionItemRow Checkbox 的关闭确认流程。
    handleDetailCloseRequest () {
      let item = null;
      let source = 'remote';

      // 当前打开的是新建会议中、尚未保存到后端的本地 Action Item。
      if (this.editingLocalIndex !== -1) {
        item = this.localActionItems[this.editingLocalIndex];
        source = 'local';
      } else {
        // actionDetailForm 是副本，必须取 action_items 内的原始引用，
        // 才能让 isPendingClose 和列表状态正确同步。
        item = this.meetingForm.action_items.find(
          action => String(action.id) === String(this.selectedActionId),
        );
      }
      if (!item || item.status === 'Closed') return;
      // 与 ActionItemRow 的 @close-request 完全复用同一个确认弹窗和确认逻辑。
      this.handleCloseRequest(item, source);
    },

    handleCloseRequest (item, source) {
      this.$set(item, 'isPendingClose', true);
      this.pendingCloseAction = { item, source };
      this.closeActionConfirmVisible = true;
    },
    cancelCloseAction () {
      if (this.closeActionLoading) return;
      if (this.pendingCloseAction && this.pendingCloseAction.item) {
        this.$set(this.pendingCloseAction.item, 'isPendingClose', false);
      }
      this.closeActionConfirmVisible = false;
      this.pendingCloseAction = null;
    },
    // Action Items (To-Do Items) 确认删除
    async confirmCloseAction () {
      if (!this.pendingCloseAction || this.closeActionLoading) return;
      const { item, source } = this.pendingCloseAction;
      if (item.status === 'Closed') {
        this.closeActionConfirmVisible = false;
        this.pendingCloseAction = null;
        return;
      }
      // 本地新增 Action Item：只更新本地状态，不调用接口。
      if (source === 'local') {
        const index = this.localActionItems.indexOf(item);
        if (index !== -1) {
          this.$set(this.localActionItems, index, {
            ...item,
            status: 'Closed',
            isPendingClose: false,
          });
        }
        this.actionDetailDialogVisible = false;
        this.closeActionConfirmVisible = false;
        this.pendingCloseAction = null;
        return;
      }
      this.closeActionLoading = true;
      try {
        // 模拟详情弹窗中勾选 Closed 后点击 Save。
        this.selectedActionId = item.id;
        this.editingLocalIndex = -1;
        this.actionDetailForm = {
          ...item,
          status: 'Closed',
          // 保留原有附件，避免 handleActionDetailSaved，将现有附件误判断为全部被删除。
          attachments: Array.isArray(item.file_list) ? item.file_list : [],
        };
        // 复用详情保存逻辑。
        await this.handleActionDetailSaved(this.actionDetailForm);
        this.closeActionConfirmVisible = false;
        this.pendingCloseAction = null;
      } catch (error) {
        console.error(error);
        this.showMessage('error', error.message || 'Failed to close action item');
      } finally {
        if (this.pendingCloseAction && this.pendingCloseAction.item) {
          this.$set(
            this.pendingCloseAction.item,
            'isPendingClose',
            false,
          );
        }
        this.closeActionLoading = false;
      }
    },
    // 显示附件数量或大小限制错误
    handleAttachmentUploadError (errors) {
      const message = Array.isArray(errors)
        ? errors.join('；')
        : String(errors);
      this.showMessage('error', message);
    },
  },
};

      const maxFiles = 10;

          rejectedFiles.push(`${file.name}（超过 100 MB）`);
          return;

        const uploadedBy = this.currentUser || 'Current User';
        this.attachmentList.push({
          _key: `new-${Date.now()}-${Math.random()}`,
          name: file.name,
          file_name: file.name,
          file,
          url: URL.createObjectURL(file),
          file_url: '',
          isLocal: true,
          uploadedBy,
          uploadedAt: new Date().toISOString(),
          isNewFile: true,
        });
      });
      if (rejectedFiles.length) this.$emit('upload-error', rejectedFiles);
    },

    deleteAttachment (attachment) {
      if (!this.hasProjectMember) return;
      if (attachment.isLocal && attachment.url) URL.revokeObjectURL(attachment.url);
      this.attachmentList = this.attachmentList.filter(
        item => item._key !== attachment._key,
      );
    },

    // 将 Action Item 中已有附件转换成列表显示所需的格式
    resetAttachments (attachments = []) {
      this.attachmentList = attachments.map((attachment, index) => {
        const name = attachment.file_name
          || attachment.name
          || 'Unnamed file';

        return {
          ...attachment,
          name,
          file_name: attachment.file_name || name,

          uploadedBy:
            attachment.upload_by
            || attachment.uploadedBy
            || '',

          url:
            attachment.file_url
            || attachment.url
            || '',

          file: attachment.file,
          isLocal: Boolean(attachment.isLocal),

          _key:
            attachment._key
            || attachment.file_id
            || attachment.id
            || `attachment-${index}-${name}`,
        };
      });
    },
    onDialogInput (val) {
      if (!val) this.close();
    },
    close () {
      this.$emit('update:visible', false);
    },
    // 查看附件；没有可直接访问的 URL 时交由父组件处理
    viewAttachment (attachment) {
      const url = attachment.url || attachment.previewUrl;
      if (!url) {
        this.$emit('view-attachment', attachment);
        return;
      }
      window.open(url, '_blank');
    },

    // 下载附件；没有可直接访问的 URL 时交由父组件处理
    async downloadAttachment (attachment) {
      const url = attachment.url || attachment.downloadUrl;
      if (!url) {
        this.$emit('download-attachment', attachment);
        return;
      }
      try {
        // 【修改】先请求文件二进制内容，避免跨域地址直接被浏览器当作预览链接打开。
        const response = await fetch(url);
        if (!response.ok) throw new Error(`Download failed: ${response.status}`);
        const blob = await response.blob();
        const blobUrl = window.URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = blobUrl;
        link.download = attachment.name || 'attachment';
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        // 释放 Blob URL，避免浏览器内存泄漏。
        window.URL.revokeObjectURL(blobUrl);
      } catch (error) {
        // 当文件服务器不允许跨域 fetch 时，仍允许用户打开文件。
        window.open(url, '_blank');
      }
    },

    getFileIcon (fileName) {
      const extension = String(fileName || '').split('.').pop().toLowerCase();
      if (extension === 'pdf') return 'mdi-file-pdf-box';
      if (['xls', 'xlsx', 'csv'].includes(extension)) return 'mdi-file-excel-box';
      if (['doc', 'docx'].includes(extension)) return 'mdi-file-word-box';
      if (['png', 'jpg', 'jpeg', 'gif', 'webp'].includes(extension)) return 'mdi-file-image-outline';
      return 'mdi-file-document-outline';
    },

    getFileIconColor (fileName) {
      const extension = String(fileName || '').split('.').pop().toLowerCase();
      if (extension === 'pdf') return '#e53935';
      if (['xls', 'xlsx', 'csv'].includes(extension)) return '#16864a';
      if (['png', 'jpg', 'jpeg', 'gif', 'webp'].includes(extension)) return '#5b8def';
      return '#64748b';
    },

    formatAttachmentDate (value) {
      const date = new Date(value);
      if (Number.isNaN(date.getTime())) return value;
      return date.toLocaleString();
    },

    handleSave () {
      if (!this.hasProjectMember || this.loading) return;
      // 仅提取本次在详情弹窗新增的原始 File 对象。
      const newFiles = this.attachmentList
        .filter(item => item.isNewFile && item.file instanceof File)
        .map(item => item.file);
      this.$emit('save', {
        ...this.form,
        attachments: this.attachmentList,
        newFiles,
      });
    },
  },
};
</script>

export default {

          return;
        }
        if (file.size > maxFileSize) {
          rejectedFiles.push(`${file.name}（超过 100 MB）`);
          return;
        }
        acceptedFiles.push(file);
      });

      acceptedFiles.forEach(file => {
        this.attachmentList.push({
          _key: `local-${Date.now()}-${Math.random()}`,
          name: file.name,
          file,
          url: URL.createObjectURL(file),
          uploadedBy: this.currentUser,
          uploadedAt: new Date().toISOString(),
          isLocal: true,
        });
      });

      // 通知父组件有文件不符合数量或大小限制
      if (rejectedFiles.length) {
        this.$emit('upload-error', rejectedFiles);
      }
    },

    // 预览附件
    viewAttachment (item) {
      const url = item.url || (item.file ? URL.createObjectURL(item.file) : '');
      if (!url) {
        this.$emit('view-attachment', item);
        return;
      }
      window.open(url, '_blank');
    },

    // 下载附件
    downloadAttachment (item) {
      const url = item.url || (item.file ? URL.createObjectURL(item.file) : '');
      if (!url) {
        this.$emit('download-attachment', item);
        return;
      }
      const link = document.createElement('a');
      link.href = url;
      link.download = item.name || 'attachment';
      link.target = '_blank';
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    },

    // 删除附件
    deleteAttachment (item) {
      if (!item.isLocal) {
        // 已存在的附件通知父组件调用后端删除接口
        this.$emit('delete-attachment', item);
      }
      if (item.isLocal && item.url) URL.revokeObjectURL(item.url);
      this.attachmentList = this.attachmentList.filter(
        attachment => attachment._key !== item._key,
      );
    },
    // 根据文件扩展名选择图标
    getFileIcon (fileName) {
      const extension = String(fileName || '')
        .split('.')
        .pop()
        .toLowerCase();
      if (extension === 'pdf') return 'mdi-file-pdf-box';
      if (['xls', 'xlsx', 'csv'].includes(extension)) return 'mdi-file-excel-box';
      if (['doc', 'docx'].includes(extension)) return 'mdi-file-word-box';
      if (['png', 'jpg', 'jpeg', 'gif', 'webp'].includes(extension)) {
        return 'mdi-file-image-outline';
      }
      return 'mdi-file-document-outline';
    },
    // 根据文件扩展名设置图标颜色
    getFileIconColor (fileName) {
      const extension = String(fileName || '')
        .split('.')
        .pop()
        .toLowerCase();
      if (extension === 'pdf') return '#e53935';
      if (['xls', 'xlsx', 'csv'].includes(extension)) return '#16864a';
      if (['png', 'jpg', 'jpeg', 'gif', 'webp'].includes(extension)) {
        return '#5b8def';
      }
      return '#64748b';
    },

    // 格式化附件上传时间
    formatAttachmentDate (value) {
      const date = new Date(value);
      if (Number.isNaN(date.getTime())) return value;
      return date.toLocaleString();
    },

    // 以下为原来的弹窗和表单逻辑
    onDialogInput (val) {
      if (!val) this.close();
    },
    close () {
      this.$emit('update:visible', false);
    },
    getAttachmentFiles () {
      return this.attachmentList
        .filter(item => item.isLocal && item.file instanceof File)
        .map(item => item.file);
    },
    // 保存数据中新增 attachments 字段
    buildSavePayload () {
      const { ownerInfo, ...rest } = this.form;
      return {
        ...(ownerInfo || {}),
        ...rest,
        status: this.form.status || 'Open',
        project_id: this.projectId,
      };
    },

    handleSave () {
      if (!this.$refs.actionForm || !this.$refs.actionForm.validate()) return;
      this.$emit('save', this.buildSavePayload(), this.getAttachmentFiles());
      this.close();
    },
  },
};
</script>

任务操作与变更上报规则:
1. 新增任务
   - 新任务通过 task-change 标记为 create。
   - 新任务继续编辑、移动、升级或降级时仍保持 create，不转为 update。
   - 新增导致其他已有任务 sort_no 变化时，受影响的已有任务标记为 update。

2. 编辑任务
   - 修改任务名称、Progress、可编辑日期，或为新建/未分配 Owner 的任务选择 Owner 时，只上报当前操作的任务。
   - 已有任务标记为 update；没有执行编辑动作的任务不需要上报。

3. 删除任务
   - 已有任务标记为 delete，父组件按任务 ID 移除其之前的 update 状态。
   - 新任务删除时只撤销 create，不调用后台删除接口。
   - 删除 Main Task 时，Main Task 和其全部 Sub Task 都必须分别标记为 delete。
   - 删除导致其他任务 sort_no 变化时，受影响的任务也需要上报。

4. 上下移动
   - 移动完成后重新生成当前任务树的 sort_no。
   - 所有 sort_no 发生变化的 Main Task 和 Sub Task 都需要上报。
   - 已有任务标记为 update，新任务仍保持 create。

5. 升级与降级
   - Sub Task 升级为 Main Task、Main Task 降级为 Sub Task 后重新生成 sort_no。
   - 被操作任务及其他 sort_no 发生变化的任务都需要上报。
   - 升级或降级只改变任务层级和顺序，不修改任务内容。

6. 状态覆盖
   - 同一个任务在父组件中最终只保留 create、update、delete 中的一种状态。
   - delete 覆盖 update；create 不会因为后续编辑或结构操作变成 update。

关键业务规则:
- Main Task 在顶层 tasks 数组中移动，并保留完整 children。
- Sub Task 只在所属 Main Task 的 children 数组中移动。
- 移动操作只改变任务顺序，不修改任务内容和父子层级。
- 组件先复制任务数据再调整结构，不直接修改 props，也不修改后端任务字段名。
- 新增任务统一先创建 Main Task；任务内容在升级或降级时原样保留。
- 日期汇总不读取 status；Sub Task 日期或任务层级变化后，通过 tasks-change 传递重新汇总的任务树。
- 最后一条 Main Task 不允许删除，避免列表清空后失去新增 Main Task 的操作入口。
- 前端新建的 Main/Sub Task 使用 isNew 标记；升级、降级和移动时保留该标记，供后续接口分流。

表头显示数据:
- tableHeaders：原表头和吸顶表头共享的显示列；日期列由 dateFields 提供，Action 仅编辑时加入。

<script>
import ProjectDatePicker from '@project_sensing/components/common/ProjectDatePicker.vue';
import ProjectInput from '@project_sensing/components/common/ProjectInput.vue';
import ProjectSelect from '@project_sensing/components/common/ProjectSelect.vue';
import TaskDetailDialog from '@project_sensing/components/workspace/scheduler/TaskDetailDialog.vue';
import { getProjectStatusConfig } from '@project_sensing/utils/projectStatus';
import { formatOwnerName } from '@project_sensing/utils/utils.js';

import moment from 'moment';

export default {
  name: 'TaskList',

  components: {
    ProjectDatePicker,
    ProjectInput,
    ProjectSelect,
    TaskDetailDialog,
  },

  props: {
    loading: {
      type: Boolean,
      default: false,
    },
    height: {
      type: Number,
      default: 0,
    },
    tasks: {
      type: Array,
      default: () => [],
    },
    members: {
      type: Array,
      default: () => [],
    },
    currentUser: {
      type: Object,
      default: () => ({}),
    },
    canManageTasks: {
      type: Boolean,
      default: false,
    },
    editMode: {
      type: Boolean,
      default: false,
    },
    requestParams: {
      type: Object,
      default: () => ({}),
    },
    projectClosed: {
      type: Boolean,
      default: false,
    },
  },

  data () {
    return {
      expandedTaskIds: [],
      detailDialogOpen: false,
      selectedDetailTask: {},
      editingTaskId: null,
      operationMode: '',
      taskNameTooltipId: null,
      dateFields: [
        { value: 'plan_start_date', text: 'Plan Start' },
        { value: 'plan_end_date', text: 'Plan End' },
        { value: 'commit_start_date', text: 'Commit Start' },
        { value: 'commit_end_date', text: 'Commit End' },
        { value: 'actual_start_date', text: 'Actual Start' },
        { value: 'actual_end_date', text: 'Actual End' },
      ],
    };
  },

  computed: {
    // 加载期间只隐藏内容行，表头保持渲染。
    renderedTasks () {
      return this.loading ? [] : this.visibleTasks;
    },
    tableHeaders () {
      const headers = [
        { value: 'task_name', text: 'Task Name' },
        { value: 'owner_name', text: 'Owner' },
        { value: 'status', text: 'Status' },
        { value: 'progress_pct', text: 'Progress' },
        ...this.dateFields,
        { value: 'detail_num', text: 'Detail' },
      ];
      return this.editMode ? [...headers, { value: 'action', text: 'Action' }] : headers;
    },
    // 将 Main/Sub 两层任务树展开为表格行，并补充层级、父任务和显示编号。
    visibleTasks () {
      const result = [];

      this.tasks.forEach((mainTask, mainIndex) => {
        result.push({
          ...mainTask,
          level: 0,
          parentId: null,
          displayNo: mainTask.sort_no || `${mainIndex + 1}.`,
        });

        if (!this.isExpanded(mainTask.id)) {
          return;
        }

        (mainTask.children || []).forEach((subTask, subIndex) => {
          result.push({
            ...subTask,
            level: 1,
            parentId: mainTask.id,
            displayNo: subTask.sort_no || `${mainIndex + 1}.${subIndex + 1}`,
          });
        });
      });

      return result;
    },
  },

  watch: {
    // 进入编辑时初始化编辑状态；空任务列表自动创建一条可填写的 Main Task。
    editMode (editing) {
      if (!editing) {
        this.editingTaskId = null;
        this.operationMode = '';
        return;
      }

      const tasks = this.cloneTasks();
      let newTask = null;
      if (!tasks.length) {
        newTask = this.createEmptyMainTask(`main-${Date.now()}`);
        tasks.push(newTask);
        this.editingTaskId = newTask.id;
        this.operationMode = 'create';
      }
      this.emitTasks(tasks);
      if (newTask) {
        this.emitTaskChange(newTask, 'create');
      }
    },

    // 首次拿到任务时默认展开全部 Main Task。
    tasks: {
      immediate: true,
      handler (tasks) {
        if (!this.expandedTaskIds.length) {
          this.expandedTaskIds = tasks.map(task => task.id);
        }
      },
    },

    // 展开状态或任务内容变化时，将当前可见行同步给 Timeline。
    visibleTasks: {
      immediate: true,
      handler (tasks) {
        this.$emit('visible-change', tasks);
      },
    },
  },

  methods: {
    formatOwnerName,
    getDetailCount (task) {
      return Number(task.detail_num) || 0;
    },
    openTaskDetail (task) {
      if (this.loading || !this.hasBackendId(task)) return;
      this.selectedDetailTask = { ...task };
      this.detailDialogOpen = true;
    },
    // 当鼠标移入 Task Name 时，判断文本是否因宽度不足被 CSS 截断。
    handleTaskNameMouseEnter (event, taskId) {
      // currentTarget 始终是绑定事件的 task-name-text，
      // 比 target 更适合用于获取自身真实的显示宽度。
      const element = event.currentTarget;
      const isTruncated = element
        && element.scrollWidth > element.clientWidth;
      this.taskNameTooltipId = isTruncated ? taskId : null;
    },

    handleTaskNameMouseLeave () {
      this.taskNameTooltipId = null;
    },
    // 深拷贝 props 任务树，所有编辑动作都基于副本处理，避免直接修改父组件数据。
    cloneTasks () {
      return JSON.parse(JSON.stringify(this.tasks));
    },

    // 根据行层级和 ID 在任务树中找到对应的 Main Task 或 Sub Task。
    findTask (tasks, task) {
      if (task.level === 0) {
        return tasks.find(item => item.id === task.id);
      }

      const parent = tasks.find(item => item.id === task.parentId);
      return parent && (parent.children || []).find(item => item.id === task.id);
    },

    // 将 Main Task 和直属 Sub Task 展平，统一检查 ID 与 sort_no。
    flattenTasks (tasks) {
      return tasks.reduce((result, mainTask) => (
        result.concat(mainTask, mainTask.children || [])
      ), []);
    },

    // 保存结构操作前的 sort_no，用于定位真正受排序影响的任务。
    getTaskSortNoMap (tasks) {
      return new Map(this.flattenTasks(tasks).map(task => [
        String(task.id),
        task.sort_no,
      ]));
    },

    // 只上报 sort_no 发生变化的任务；新任务通过 isNew 继续保持 create。
    emitSortNoChanges (tasks, beforeSortNoMap) {
      this.flattenTasks(tasks).forEach(task => {
        const taskId = String(task.id);
        if (beforeSortNoMap.has(taskId)
          && beforeSortNoMap.get(taskId) !== task.sort_no) {
          this.emitTaskChange(task, 'update');
        }
      });
    },

    // 后台删除 Main Task 时要求同时传入该 Main Task 下全部 Sub Task 的 ID。
    getDeletedTasks (task) {
      return task.level === 0
        ? [task, ...(task.children || [])]
        : [task];
    },

    // 删除覆盖已有状态；新增任务后续编辑仍保持 create。
    getTaskActionStatus (task, type) {
      if (type === 'delete') return 'delete';
      if (type === 'create' || task.isNew || task.actionStatus === 'create') return 'create';
      return 'update';
    },

    // 只发送本次新增、修改或删除的 Task；新任务继续编辑时仍保持 create。
    emitTaskChange (task, type) {
      task.actionStatus = this.getTaskActionStatus(task, type);
      this.$emit('task-change', {
        type: task.actionStatus,
        task: JSON.parse(JSON.stringify(task)),
      });
    },

    // 统一整理层级编号和 Main Task 汇总日期，再将完整任务树传给父组件。
    emitTasks (tasks) {
      const normalizedTasks = this.normalizeTaskMeta(tasks);
      const nextTasks = this.aggregateMainTaskDates(normalizedTasks);
      this.$emit('tasks-change', nextTasks);
      return nextTasks;
    },

    // 按当前父子顺序重新生成 task_type 和后端展示编号 sort_no。
    normalizeTaskMeta (tasks) {
      return tasks.map((mainTask, mainIndex) => {
        mainTask.task_type = 'Master';
        mainTask.sort_no = `${mainIndex + 1}.`;
        (mainTask.children || []).forEach((subTask, subIndex) => {
          subTask.task_type = 'Sub';
          subTask.sort_no = `${mainIndex + 1}.${subIndex + 1}`;
        });
        return mainTask;
      });
    },

    // 有 Sub Task 时，使用子任务最早 Start、最晚 End 汇总 Main Task 日期。
    aggregateMainTaskDates (tasks) {
      return tasks.map(mainTask => {
        const children = Array.isArray(mainTask.children)
          ? mainTask.children
          : [];

        if (!children.length) {
          return mainTask;
        }

        const aggregateDate = (field, mode) => {
          const dates = children
            .map(child => child[field])
            .filter(date => moment(date, 'YYYY-MM-DD', true).isValid());

          if (!dates.length) {
            return '';
          }

          return dates.sort()[mode === 'min' ? 0 : dates.length - 1];
        };

        mainTask.plan_start_date = aggregateDate('plan_start_date', 'min');
        mainTask.plan_end_date = aggregateDate('plan_end_date', 'max');
        mainTask.commit_start_date = aggregateDate('commit_start_date', 'min');
        mainTask.commit_end_date = aggregateDate('commit_end_date', 'max');
        mainTask.actual_start_date = aggregateDate('actual_start_date', 'min');
        // 不依赖后台状态；只有每个 Sub Task 都填写 Actual End，Main Task 才产生 Actual End。
        mainTask.actual_end_date = children.every(child => (
          moment(child.actual_end_date, 'YYYY-MM-DD', true).isValid()
        ))
          ? aggregateDate('actual_end_date', 'max')
          : '';

        return mainTask;
      });
    },

    // 更新任务普通字段，并通过统一出口重新汇总和上报任务树。
    updateField (task, field, value) {
      if (!this.canUpdateField(task, field)) return;
      const tasks = this.cloneTasks();
      const target = this.findTask(tasks, task);

      if (!target) {
        return;
      }

      target[field] = value;
      this.emitTasks(tasks);
      this.emitTaskChange(target, 'update');
    },

    // 根据下拉选择同时写入保存需要的 owner_id 和 owner_name。
    updateOwner (task, ownerId) {
      if (!this.canEditOwner(task)) return;
      const member = this.members.find(item => item.owner_id === ownerId);
      const tasks = this.cloneTasks();
      const target = this.findTask(tasks, task);

      if (!target) {
        return;
      }

      target.owner_id = ownerId;
      target.owner_name = member ? member.owner_name : '';
      this.emitTasks(tasks);
      this.emitTaskChange(target, 'update');
    },

    // 将进度转换为 0 至 100 的数字后更新 progress_pct。
    updateProgress (task, value) {
      const progress = Math.min(100, Math.max(0, Number(value) || 0));
      this.updateField(task, 'progress_pct', progress);
    },

    // 新任务直接编辑；已有任务只有点击铅笔后才进入行编辑状态。
    isTaskEditing (task) {
      return this.editMode
        && !this.projectClosed
        && (task.isNew || this.editingTaskId === task.id);
    },

    // Sub Task 继承所属 Main Task Owner 的管理权限；不改变项目级 GAC 权限。
    isMainTaskOwner (task) {
      if (task.level !== 1) return false;
      const parent = this.tasks.find(item => String(item.id) === String(task.parentId));
      return Boolean(parent) && this.isTaskOwner(parent);
    },

    // 字段权限集中在 TaskList，避免影响 Project、Meeting 等共用权限。
    canEditTaskField (task, field) {
      if (!this.editMode || this.projectClosed) return false;
      const isManager = this.canManageTasks;
      const isOwner = this.isTaskOwner(task);
      const isParentOwner = this.isMainTaskOwner(task);

      if (field === 'task_name' || field === 'owner_id') {
        return isManager || (task.level === 1 && isParentOwner);
      }
      if (field === 'progress_pct') {
        return isManager || isOwner || isParentOwner;
      }
      if (['plan_start_date', 'plan_end_date'].includes(field)) {
        return isManager || (task.level === 1 && (isParentOwner || isOwner));
      }
      if (['commit_start_date', 'commit_end_date', 'actual_start_date', 'actual_end_date'].includes(field)) {
        return isManager || isOwner || isParentOwner;
      }
      return false;
    },

    // 保留原有 Commit 日期填写后锁定 Owner 的业务规则。
    canEditOwner (task) {
      return this.isTaskEditing(task)
        && this.canEditTaskField(task, 'owner_id')
        && !String(task.commit_start_date || '').trim()
        && !String(task.commit_end_date || '').trim();
    },

    // Main Task 只能由项目管理员新增；Sub Task 允许所属 Main Task Owner 新增。
    canAddTask (task) {
      if (!this.editMode || this.projectClosed) return false;
      return this.canManageTasks || (task.level === 1 && this.isMainTaskOwner(task));
    },

    // 判断任务是否为带后台 ID 的已保存数据。
    hasBackendId (task) {
      return !task.isNew
        && task.id !== undefined
        && task.id !== null
        && task.id !== '';
    },

    // 新增期间禁止进入已有任务的编辑模式，避免新增和编辑混合保存。
    canEditTask (task) {
      return this.operationMode !== 'create'
        && this.hasBackendId(task)
        && (this.canManageTasks || this.isTaskOwner(task) || this.isMainTaskOwner(task));
    },

    // 新增期间只允许操作新任务，已有任务的结构操作保持禁用。
    canOperateTask (task) {
      return this.operationMode !== 'create' || task.isNew;
    },

    // 将已有任务切换为 update 模式并打开该行输入控件。
    editTask (task) {
      if (!this.canEditTask(task)) {
        return;
      }

      this.operationMode = 'update';
      this.editingTaskId = task.id;
    },
    hasCompletePlanDates (task) {
      return Boolean(String(task.plan_start_date || '').trim())
        && Boolean(String(task.plan_end_date || '').trim());
    },
    hasCompleteCommitDates (task) {
      return Boolean(String(task.commit_start_date || '').trim())
        && Boolean(String(task.commit_end_date || '').trim());
    },
    // 日期原有前置条件和 Main Task 汇总锁定规则保持不变，仅调整操作人权限。
    canEditDate (task, field) {
      const hasChildren = task.level === 0
        && Array.isArray(task.children)
        && task.children.length > 0;
      if (!this.canEditTaskField(task, field) || hasChildren) return false;
      if (['commit_start_date', 'commit_end_date'].includes(field)) {
        return this.hasCompletePlanDates(task);
      }
      if (['actual_start_date', 'actual_end_date'].includes(field)) {
        return this.hasCompleteCommitDates(task);
      }
      return true;
    },

    // 工号统一转成大写后比较；Admin/PM 非 Owner 时也不能填写 Commit 时间。
    isTaskOwner (task) {
      const currentUserId = String(this.currentUser.id || '').trim().toUpperCase();
      return Boolean(currentUserId)
        && String(task.owner_id || '').trim().toUpperCase() === currentUserId;
    },

    // 方法内部也验证权限；日期仍复用原有的可编辑条件。
    canUpdateField (task, field) {
      if (!this.isTaskEditing(task)) return false;
      if (this.dateFields.some(item => item.value === field)) {
        return this.canEditDate(task, field);
      }
      return this.canEditTaskField(task, field);
    },
    // End 日期选择器最小值取同组 Start，确保 Start 不晚于 End。
    getDateMin (task, field) {
      const startFieldMap = {
        plan_end_date: 'plan_start_date',
        commit_end_date: 'commit_start_date',
        actual_end_date: 'actual_start_date',
      };
      return task[startFieldMap[field]] || '';
    },

    // Start 日期选择器最大值取同组 End，确保 Start 不晚于 End。
    getDateMax (task, field) {
      const endFieldMap = {
        plan_start_date: 'plan_end_date',
        commit_start_date: 'commit_end_date',
        actual_start_date: 'actual_end_date',
      };
      return task[endFieldMap[field]] || '';
    },

    // 在当前 Main Task 后插入一条空 Main Task，并切换为 create 模式。
    addMainTask (task) {
      if (!this.canManageTasks || this.operationMode === 'update') {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const index = tasks.findIndex(item => item.id === task.id);
      if (index === -1) {
        return;
      }

      const newTask = this.createEmptyMainTask(`main-${Date.now()}`);
      tasks.splice(index + 1, 0, newTask);
      this.operationMode = 'create';
      this.editingTaskId = newTask.id;
      const nextTasks = this.emitTasks(tasks);
      this.emitTaskChange(newTask, 'create');
      this.emitSortNoChanges(nextTasks, beforeSortNoMap);
    },

    // 在当前 Sub Task 后新增一条同级 Sub Task，并保持在原 Main Task 的 children 中。
    addSubTask (task) {
      if (!this.canAddTask(task) || this.operationMode === 'update' || task.level !== 1) {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const parent = tasks.find(item => item.id === task.parentId);
      const subTaskIndex = parent
        ? (parent.children || []).findIndex(item => item.id === task.id)
        : -1;
      if (!parent || subTaskIndex === -1) {
        return;
      }

      const newTask = this.createEmptySubTask(`sub-${Date.now()}`);
      parent.children.splice(subTaskIndex + 1, 0, newTask);
      this.operationMode = 'create';
      this.editingTaskId = newTask.id;
      const nextTasks = this.emitTasks(tasks);
      this.emitTaskChange(newTask, 'create');
      this.emitSortNoChanges(nextTasks, beforeSortNoMap);
    },

    // 复用空 Sub Task 字段创建 Main Task，并增加 children 数组。
    createEmptyMainTask (id) {
      return {
        ...this.createEmptySubTask(id),
        children: [],
      };
    },

    // 创建使用后端字段名的空任务，新任务通过 isNew 参与保存分流。
    createEmptySubTask (id) {
      return {
        id,
        isNew: true,
        task_type: 'Sub',
        sort_no: '',
        task_name: '',
        owner_id: '',
        owner_name: '',
        status: '',
        progress_pct: 0,
        plan_start_date: '',
        plan_end_date: '',
        commit_start_date: '',
        commit_end_date: '',
        actual_start_date: '',
        actual_end_date: '',
        detail_num: 0,
      };
    },

    // 后台为空时自动生成的第一条新 Main Task 使用 Reset，清空内容并撤销此前的 create 记录。
    resetTask (task) {
      if (!this.canManageTasks || !this.isResettableTask(task)) {
        return;
      }

      const resetTasks = this.getDeletedTasks(task);
      const emptyTask = this.createEmptyMainTask(task.id);
      this.editingTaskId = emptyTask.id;
      this.operationMode = 'create';
      this.expandedTaskIds = [emptyTask.id];
      this.emitTasks([emptyTask]);
      resetTasks.forEach(resetTask => this.emitTaskChange(resetTask, 'delete'));
    },
    // 删除指定任务；删除 Main Task 时一并移除其 children。
    deleteTask (task) {
      if (!this.canDeleteTask(task)) {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const deletedTasks = this.getDeletedTasks(task);
      const deletedTaskIds = new Set(deletedTasks.map(item => String(item.id)));

      // 删除当前编辑任务或其所属 Main Task 时，清除行编辑状态，避免其他新增按钮持续禁用。
      if (deletedTaskIds.has(String(this.editingTaskId))) {
        this.editingTaskId = null;
        this.operationMode = '';
      }
      // 展开状态只保留仍存在的 Main Task，避免删除后残留无效 ID。
      this.expandedTaskIds = this.expandedTaskIds.filter(id => (
        !deletedTaskIds.has(String(id))
      ));

      if (task.level === 0) {
        const nextTasks = tasks.filter(item => item.id !== task.id);
        this.resetCreateModeIfNeeded(nextTasks);
        const normalizedTasks = this.emitTasks(nextTasks);
        deletedTasks.forEach(deletedTask => this.emitTaskChange(deletedTask, 'delete'));
        this.emitSortNoChanges(normalizedTasks, beforeSortNoMap);
        return;
      }

      const parent = tasks.find(item => item.id === task.parentId);
      if (parent) {
        parent.children = (parent.children || []).filter(item => item.id !== task.id);
        this.resetCreateModeIfNeeded(tasks);
        const normalizedTasks = this.emitTasks(tasks);
        deletedTasks.forEach(deletedTask => this.emitTaskChange(deletedTask, 'delete'));
        this.emitSortNoChanges(normalizedTasks, beforeSortNoMap);
      }
    },

    // Sub Task 均可删除；Main Task 至少保留一条作为继续新增的入口。
    canDeleteTask (task) {
      if (!this.editMode || this.projectClosed) return false;
      const hasPermission = this.canManageTasks
        || (task.level === 1 && this.isMainTaskOwner(task));
      return hasPermission && (task.level === 1 || this.tasks.length > 1);
    },

    // 只有后台为空时自动生成的第一条新 Main Task 才显示 Reset；已有后台 ID 的任务保持 Delete。
    isResettableTask (task) {
      return task.level === 0
        && task.isNew
        && !this.hasBackendId(task)
        && this.tasks.length === 1;
    },
    // 当任务树已无 isNew 数据时退出 create 模式，恢复已有任务操作。
    resetCreateModeIfNeeded (tasks) {
      const hasNewTask = tasks.some(mainTask => (
        mainTask.isNew
        || (mainTask.children || []).some(subTask => subTask.isNew)
      ));

      if (this.operationMode === 'create' && !hasNewTask) {
        this.operationMode = '';
        this.editingTaskId = null;
      }
    },

    // 获取任务同层级数组，Main 使用顶层 tasks，Sub 使用所属 Main 的 children。
    getTaskSiblings (tasks, task) {
      if (task.level === 0) {
        return tasks;
      }

      const parent = tasks.find(item => item.id === task.parentId);
      return parent ? parent.children || [] : [];
    },

    // 判断目标方向是否仍在当前同级数组范围内。
    canMove (task, offset) {
      const siblings = this.getTaskSiblings(this.tasks, task);
      const index = siblings.findIndex(item => item.id === task.id);
      const targetIndex = index + offset;
      return index !== -1 && targetIndex >= 0 && targetIndex < siblings.length;
    },

    // 在同级数组中交换任务顺序，不改变 Main/Sub 层级关系。
    moveTask (task, offset) {
      if (!this.canOperateTask(task)) {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const siblings = this.getTaskSiblings(tasks, task);
      const index = siblings.findIndex(item => item.id === task.id);
      const targetIndex = index + offset;

      if (index === -1 || targetIndex < 0 || targetIndex >= siblings.length) {
        return;
      }

      // 上下移动只调整当前层级内的顺序，不改变 Main Task 与 Sub Task 的归属关系。
      const movedTask = siblings.splice(index, 1)[0];
      siblings.splice(targetIndex, 0, movedTask);
      const nextTasks = this.emitTasks(tasks);
      this.emitSortNoChanges(nextTasks, beforeSortNoMap);
    },

    // 只有 Sub Task 可以升级为 Main Task。
    canPromote (task) {
      return task.level === 1;
    },

    // 将 Sub Task 从原 children 移出，并插入原 Main Task 的下一行。
    promoteTask (task) {
      if (!this.canOperateTask(task)) {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const parentIndex = tasks.findIndex(item => item.id === task.parentId);

      if (parentIndex === -1) {
        return;
      }

      const children = tasks[parentIndex].children || [];
      const childIndex = children.findIndex(item => item.id === task.id);

      if (childIndex === -1) {
        return;
      }

      // 升级只改变层级，任务字段原样保留。
      const promotedTask = children.splice(childIndex, 1)[0];
      promotedTask.children = [];
      tasks.splice(parentIndex + 1, 0, promotedTask);
      const nextTasks = this.emitTasks(tasks);
      this.emitSortNoChanges(nextTasks, beforeSortNoMap);
    },

    // 非第一条且没有 children 的 Main Task 才能降级。
    canDemote (task) {
      if (task.level !== 0 || (task.children || []).length) {
        return false;
      }

      return this.tasks.findIndex(item => item.id === task.id) > 0;
    },

    // 将 Main Task 降级并追加到前一个 Main Task 的 children 末尾。
    demoteTask (task) {
      if (!this.canOperateTask(task)) {
        return;
      }

      const tasks = this.cloneTasks();
      const beforeSortNoMap = this.getTaskSortNoMap(tasks);
      const taskIndex = tasks.findIndex(item => item.id === task.id);

      if (taskIndex <= 0 || (tasks[taskIndex].children || []).length) {
        return;
      }

      const parent = tasks[taskIndex - 1];
      const demotedTask = tasks.splice(taskIndex, 1)[0];

      if (!Array.isArray(parent.children)) {
        this.$set(parent, 'children', []);
      }

      // 降级任务归入前一个 Main Task，并放在其现有 Sub Task 的最后。
      delete demotedTask.children;
      parent.children.push(demotedTask);

      if (!this.isExpanded(parent.id)) {
        this.expandedTaskIds.push(parent.id);
      }

      const nextTasks = this.emitTasks(tasks);
      this.emitSortNoChanges(nextTasks, beforeSortNoMap);
    },

    // 判断 Main Task 当前是否处于展开状态。
    isExpanded (id) {
      return this.expandedTaskIds.includes(id);
    },

    // 切换 Main Task 展开状态，仅影响可见行，不修改任务树。
    toggleTask (task) {
      const index = this.expandedTaskIds.indexOf(task.id);
      if (index === -1) {
        this.expandedTaskIds.push(task.id);
        return;
      }
      this.expandedTaskIds.splice(index, 1);
    },

    // 将空值或非法进度显示为 0，同时保留正常数字。
    normalizeProgress (progress) {
      return Number(progress) || 0;
    },

    // 获取任务状态对应的统一颜色配置。
    getStatusConfig (status) {
      return getProjectStatusConfig(status);
    },

    // Actual 日期超出所属 Main Task Plan 范围时，Actual Start / End 同时标红。
    // 仅判断显示颜色，不修改日期、编辑权限或自动汇总规则。
    isActualDateOutOfPlan (task, field) {
      if (!['actual_start_date', 'actual_end_date'].includes(field)) return false;

      const mainTask = task.level === 0
        ? task
        : this.tasks.find(item => String(item.id) === String(task.parentId));
      if (!mainTask) return false;

      const parseDate = value => moment(value, ['YYYY-MM-DD', 'YYYY/MM/DD'], true);
      const planStart = parseDate(mainTask.plan_start_date);
      const planEnd = parseDate(mainTask.plan_end_date);
      if (!planStart.isValid() || !planEnd.isValid()) return false;

      const actualStart = parseDate(task.actual_start_date);
      const actualEnd = parseDate(task.actual_end_date);
      return [actualStart, actualEnd].some(date => (
        date.isValid() && (date.isBefore(planStart, 'day') || date.isAfter(planEnd, 'day'))
      ));
    },

    // 浏览模式将完整日期格式化为 MM-DD，空日期显示短横线。
    formatDate (date) {
      return date ? moment(date, 'YYYY-MM-DD').format('YYYY/MM/DD') : '-';
    },
    getOwnerDisplayData (task) {
      const member = this.members.find(item => (
        String(item.owner_id || '').trim().toUpperCase() === String(task.owner_id || '').trim().toUpperCase()
      ));

      const ownerName = String(
        task.owner_name
        || (member && member.owner_name)
        || '',
      ).trim();

      const chineseName = String(
        task.owner_chinese_name
        || (member && member.owner_chinese_name)
        || '',
      ).trim();

      const englishName = String(
        task.owner_english_name
        || (member && member.owner_english_name)
        || '',
      ).trim();

      const employeeId = String(
        task.owner_id
        || (member && member.owner_id)
        || '',
      ).trim();

      // 只有中文字段中确实包含中文字符时，才判断为国内用户。
      const isChineseUser = /[\u4e00-\u9fff]/.test(
        `${chineseName}${ownerName}`,
      );

      return {
        ownerName,
        chineseName,
        englishName,
        employeeId,
        isChineseUser,
      };
    },

    taskFormatOwnerName (ownerName) {
      const name = String(ownerName || '').trim();
      if (!name) return '-';

      // 中文、数字或其他非纯英文姓名保持完整显示
      if (!/^[A-Za-z\s'-]+$/.test(name)) return name;

      const nameParts = name.split(/\s+/).filter(Boolean);

      // 超过两个英文单词时，只显示前两个单词
      return nameParts.length > 2
        ? `${nameParts.slice(0, 2).join(' ')}...`
        : name;
    },

    // 表格默认显示
    formatOwnerDisplay(task) {
      const {
        ownerName,
        chineseName,
        englishName,
        employeeId,
        isChineseUser,
      } = this.getOwnerDisplayData(task);

      if (isChineseUser) {
        const displayChineseName = chineseName || ownerName;

        // 中文用户：中文名（英文名）
        if (displayChineseName && englishName) {
          return `${displayChineseName} (${englishName})`;
        }

        return displayChineseName || employeeId || '-';
      }

      // 外国用户：只显示截断后的英文名，不显示工号
      const foreignName = englishName || ownerName;

      return foreignName
        ? this.taskFormatOwnerName(foreignName)
        : '-';
    },

    // Tooltip 显示完整内容
    formatOwnerFullDisplay (task) {
      const {
        ownerName,
        chineseName,
        englishName,
        employeeId,
        isChineseUser,
      } = this.getOwnerDisplayData(task);

      if (isChineseUser) {
        const displayChineseName = chineseName || ownerName;

        // 中文用户：Tooltip 显示 中文名（英文名，工号）
        if (displayChineseName && englishName) {
          return `${displayChineseName}（${englishName}，${employeeId}）`;
        }

        return displayChineseName
          ? `${displayChineseName}（${employeeId}）`
          : employeeId || '-';
      }

      // 外国用户：Tooltip 显示 英文名（工号）
      const foreignName = englishName || ownerName;

      return foreignName
        ? `${foreignName}（${employeeId}）`
        : employeeId || '-';
    },

    // 判断是否需要显示 Tooltip
    hasTruncatedOwnerName (task) {
      const {
        ownerName,
        englishName,
        isChineseUser,
      } = this.getOwnerDisplayData(task);

      // 国内用户：有英文名时，Tooltip 展示完整中文名和英文名
      if (isChineseUser) {
        return Boolean(englishName);
      }

      // 国外用户：英文名超过两个单词时显示 Tooltip
      const foreignName = String(
        englishName || ownerName || '',
      ).trim();

      if (!/^[A-Za-z\s'-]+$/.test(foreignName)) {
        return false;
      }

      const nameParts = foreignName.split(/\s+/).filter(Boolean);

      return nameParts.length > 2;
    },
  },
};
</script>
